## 1. Ⅴ个创建型模式

这些设计模式提供了一种在创建对象的同时隐藏创建逻辑的方式，而不是使用 new 运算符直接实例化对象。

### 1.1 工厂方法模式（Factory Method Pattern）

工厂模式的核心理念是：以单一方式创建继承同一基类\接口的不同类型对象。

```cpp
#include <iostream>
// 枚举
enum ShapeType{
    CIRCLE,
    RECTANGLE
};
// 接口类
class Shape{
public:
    virtual void draw() = 0;
    virtual ~Shape() = default;
};
// 具体类
class Circle: public Shape{
    public :
    void draw() override{
        std::cout << "Circle" << std::endl;
    }
};
class Rectangle: public Shape{
    public :
    void draw() override{
        std::cout << "Rectangle" << std::endl;
    }
};
// 工厂类
class Factory{
public:
    static Shape* createShape(const ShapeType type){
        switch (type)
        {
        case CIRCLE:
            return new Circle();
            break;
        case RECTANGLE:
            return new Rectangle();
            break;
        default:
            return nullptr;
            break;
        }
    }
};
// 客户端代码
int main(int argc, char **argv)
{
    Shape* shape1 = Factory::createShape(CIRCLE);
    shape1->draw();
    delete shape1;
    Shape* shape2 = Factory::createShape(RECTANGLE);
    shape2->draw();
    delete shape2;

    return 0;
}

```

调用者并不知道运行时真正的类名，所以对工厂进行扩展的方式（防止程序员调用出错）可以考虑使用一个枚举类（防止传入参数时，把circle拼写错误）。

### 1.2 抽象工厂模式（Abstract Factory Pattern）

抽象工厂模式与工厂模式类似，只不过它通过间接创建工厂分族创建一系列相关产品。

```cpp
#include <iostream>
// 枚举
enum FactoryType{
    SHAPE_FACTORY,
    COLOR_FACTORY
};
enum ColorType{
    RED,
    GREEN,
    BLUE
};
enum ShapeType{
    CIRCLE,
    RECTANGLE
};
// 接口类
class Shape{
public:
    virtual void draw() = 0;
    virtual ~Shape() = default;
};
class Color{
public:
    virtual void fill() = 0;
    virtual ~Color() = default;
};
// 具体类
class Circle: public Shape{
    public :
    void draw() override{
        std::cout << "Circle" << std::endl;
    }
};
class Rectangle: public Shape{
    public :
    void draw() override{
        std::cout << "Rectangle" << std::endl;
    }
};
class Red: public Color{
    public :
    void fill() override{
        std::cout << "Red" << std::endl;
    }
};
class Green: public Color{
    public :
    void fill() override{
        std::cout << "Green" << std::endl;
    }
};
//  抽象工厂类
class AbstractFactory{
public:
    virtual Shape* createShape(const ShapeType type) = 0;
    virtual Color* createColor(const ColorType type) = 0;
    virtual ~AbstractFactory() = default;
};
// 工厂类
class ShapeFactory: public AbstractFactory{
public: 
    Shape* createShape(const ShapeType type) override{
        switch (type)
        {
        case CIRCLE:
            return new Circle();
            break;
        case RECTANGLE:
            return new Rectangle();
            break;
        default:
            return nullptr;
            break;
        }
    }
    Color* createColor(const ColorType type) override{
        return nullptr;
    }
};
class ColorFactory: public AbstractFactory{
public: 
    Shape* createShape(const ShapeType type) override{
        return nullptr;
    }
    Color* createColor(const ColorType type) override{
        switch (type)
        {
        case RED:
            return new Red();
            break;
        case GREEN:
            return new Green();
            break;
        default:
            return nullptr;
            break;
        }
    }
};
// 工厂生成器类
class FactoryProducer{
public:
    static AbstractFactory* getFactory(const FactoryType type){
        switch (type)
        {
        case SHAPE_FACTORY:
            return new ShapeFactory();
            break;
        case COLOR_FACTORY:
            return new ColorFactory();
            break;
        default:
            return nullptr;
            break;
        }
    }
};
// 客户端代码
int main(int argc, char **argv)
{
    AbstractFactory* shapeFactory = FactoryProducer::getFactory(SHAPE_FACTORY);
    Shape* shape1 = shapeFactory->createShape(CIRCLE);
    shape1->draw();
    Shape* shape2 = shapeFactory->createShape(RECTANGLE);
    shape2->draw();
    delete shape1;
    delete shape2;
    delete shapeFactory;
    AbstractFactory* colorFactory = FactoryProducer::getFactory(COLOR_FACTORY);
    Color* color1 = colorFactory->createColor(RED);
    color1->fill();
    Color* color2 = colorFactory->createColor(GREEN);
    color2->fill();
    delete color1;
    delete color2;
    delete colorFactory;

    return 0;
}
```

### 1.3 单例模式（Singleton Pattern）

单例模式：确保每一个类只有一个实例，从而减少频繁创建销毁类实例的性能消耗。

```cpp
// 饿汉式
class Singleton {
public:
    static Singleton& instance() {          // 返回引用，杜绝拷贝
        return inst;
    }
    // 删除拷贝/移动，彻底禁止复制
    Singleton(Singleton&&)                 = delete;
    Singleton& operator=(Singleton&&)      = delete;

private:
    Singleton()  = default;                 // 私有构造
    ~Singleton() = default;                 // 私有析构（也可公开，视需要）
    static Singleton inst;
};

// 懒汉式--线程不安全
class Singleton2{
public:
    static Singleton2 * instance() {          // 返回引用，杜绝拷贝
        static Singleton2 * p = nullptr;
        if(!p) p = new Singleton2;
        return p;
    }
    // 删除拷贝/移动，彻底禁止复制
    Singleton2(Singleton2&&)                 = delete;
    Singleton2& operator=(Singleton2&&)      = delete;
    private:
    Singleton2()  = default;                 // 私有构造
    ~Singleton2() = default;                 // 私有析构（也可公开，视需要）
};

// 懒汉式--双检锁DCLP（C++11前使用）
class Singleton3{
public:
    static Singleton3 * instance() {          // 返回引用，杜绝拷贝
        static std::mutex mtx;
        static Singleton3 * p = nullptr;
        if(!p) {
            std::lock_guard<std::mutex> lock(mtx);
            if(!p) p = new Singleton3;
        }
        return p;
    }
    // 删除拷贝/移动，彻底禁止复制
    Singleton3(Singleton3&&)                 = delete;
    Singleton3& operator=(Singleton3&&)      = delete;
    private:
    Singleton3()  = default;                 // 私有构造
    ~Singleton3() = default;                 // 私有析构（也可公开，视需要）
  
};

// 饿汉式静态成员初始化（C++11后推荐）
class Singleton4{
public:
    static Singleton4& instance() {          // 返回引用，杜绝拷贝
        static Singleton4 inst;            // C++11后局部静态变量初始化是线程安全的
        return inst;
    }
    // 删除拷贝/移动，彻底禁止复制
    Singleton4(Singleton4&&)                 = delete;
    Singleton4& operator=(Singleton4&&)      = delete;
    private:
    Singleton4()  = default;                 // 私有构造
    ~Singleton4() = default;                 // 私有析构（也可公开，视需要）
};

// 可继承单例
template <class T>
class SingletonBase{
    public:
    static D& instance() {
        static D inst;
        return inst;
    }
protected:
    SingletonBase() = default;
    ~SingletonBase() = default;
private:
    SingletonBase(const SingletonBase&) = delete;
    void operator=(const SingletonBase&) = delete;
};

class Logger : public SingletonBase<Logger> {
    friend class SingletonBase<Logger>;
    Logger() = default;
};
```

### 1.4 建造者模式（Builder Pattern）

建造者模式与工厂模式相似，但是分阶段装配产品，而不是像工厂模式一样一条语句产出产品，适合复杂对象生成。

```cpp
// 产品类
class GameConsole {
public:
    void setCpu(std::string v)      { cpu_ = std::move(v); }
    void setGpu(std::string v)      { gpu_ = std::move(v); }
    void setRam(std::string v)      { ram_ = std::move(v); }
    void setSsd(std::string v)      { ssd_ = std::move(v); }
    void show() const {
        std::cout << "Console: " << cpu_ << ' ' << gpu_ << ' '
                  << ram_ << ' ' << ssd_ << "\n\n";
    }
private:
    std::string cpu_, gpu_, ram_, ssd_;
};
// 抽象建造者
class IBuilder {
public:
    virtual ~IBuilder() = default;
    virtual void buildCpu() = 0;
    virtual void buildGpu() = 0;
    virtual void buildRam() = 0;
    virtual void buildSsd() = 0;
    virtual GameConsole getResult() = 0;
};
// 具体建造者
class SwitchBuilder : public IBuilder {
public:
    void buildCpu() override { console_.setCpu("ARM Cortex-A57"); }
    void buildGpu() override { console_.setGpu("NVIDIA Maxwell"); }
    void buildRam() override { console_.setRam("4 GB LPDDR4"); }
    void buildSsd() override { console_.setSsd("32 GB eMMC"); }
    GameConsole getResult() override { return std::move(console_); }
private:
    GameConsole console_;
};

class PS5Builder : public IBuilder {
public:
    void buildCpu() override { console_.setCpu("AMD Zen2 8-core"); }
    void buildGpu() override { console_.setGpu("AMD RDNA2"); }
    void buildRam() override { console_.setRam("16 GB GDDR6"); }
    void buildSsd() override { console_.setSsd("825 GB NVMe"); }
    GameConsole getResult() override { return std::move(console_); }
private:
    GameConsole console_;
};
// 指挥者
class Director {
public:
    GameConsole make(IBuilder& builder) {
        builder.buildCpu();
        builder.buildGpu();
        builder.buildRam();
        builder.buildSsd();
        return builder.getResult();
    }
};
// 客户端代码
int main() {
    Director dir;

    SwitchBuilder swB;
    auto switchConsole = dir.make(swB);
    switchConsole.show();

    PS5Builder psB;
    auto ps5Console = dir.make(psB);
    ps5Console.show();
}
```

### 1.5 原型模式（Prototype Pattern）

这种模式是实现了一个原型接口，该接口用于创建当前对象的克隆。当直接创建对象的代价比较大时，则采用这种模式。例如，一个对象需要在一个高代价的数据库操作之后被创建。我们可以缓存该对象，在下一个请求时返回它的克隆，在需要的时候更新数据库，以此来减少数据库调用。

```cpp
// 抽象原型
class Monster {
public:
    virtual ~Monster() = default;
    virtual std::unique_ptr<Monster> clone() const = 0; // 原型核心
    virtual void info() const = 0;
    void setPos(int x, int y) { x_ = x; y_ = y; }
protected:
    int x_{}, y_{};
    std::string name_;
};
// 具体原型
class Orc : public Monster {
public:
    Orc() { name_ = "Orc"; }
    std::unique_ptr<Monster> clone() const override {
        return std::make_unique<Orc>(*this); // 拷贝构造=深拷贝
    }
    void info() const override {
        std::cout << name_ << " at (" << x_ << ',' << y_ << ")\n";
    }
};

class Dragon : public Monster {
public:
    Dragon() { name_ = "Dragon"; }
    std::unique_ptr<Monster> clone() const override {
        return std::make_unique<Dragon>(*this);
    }
    void info() const override {
        std::cout << name_ << " at (" << x_ << ',' << y_ << ")\n";
    }
};
// 原型管理器
class MonsterFactory {
public:
    using PrototypeCtor = std::function<std::unique_ptr<Monster>()>;

    static MonsterFactory& inst() {
        static MonsterFactory f;
        return f;
    }
    // 注册原型至原型图表
    void registerPrototype(const std::string& key, PrototypeCtor ctor) {
        prototypes_[key] = std::move(ctor);
    }
    // 原型图表中查找原型并克隆一个
    std::unique_ptr<Monster> spawn(const std::string& key) {
        auto it = prototypes_.find(key);
        if (it == prototypes_.end()) return nullptr;
        return it->second()->clone(); // 先拿原型，再克隆
    }

private:
    // 原型图表
    std::unordered_map<std::string, PrototypeCtor> prototypes_;
};
// 客户端代码
int main() {
   // 单例
    auto& factory = MonsterFactory::inst();

    // 注册原型（通常放在程序启动处）
    factory.registerPrototype("Orc", []() { return std::make_unique<Orc>(); });
    factory.registerPrototype("Dragon", []() { return std::make_unique<Dragon>(); });

    // 刷怪——完全解耦
    auto m1 = factory.spawn("Orc");
    auto m2 = factory.spawn("Dragon");
    if (m1) { m1->setPos(10, 20); m1->info(); }
    if (m2) { m2->setPos(30, 40); m2->info(); }
}
```

## 2. Ⅷ个结构型模式

### 2.1 适配器模式（Adapter Pattern）

适配器模式（Adapter Pattern）充当两个不兼容接口之间的桥梁，属于结构型设计模式。它通过一个中间件（适配器）将一个类的接口转换成客户期望的另一个接口，使原本不能一起工作的类能够协同工作。

更简单地说，适配器模式就是将原本不兼容的旧的接口"包装"成新类，新类使用旧的方法并兼容其他方法。

对象适配器（推荐，组合代替继承）

```cpp
// 目标接口（新系统）
class ModernRenderer {
public:
    virtual void drawRoundedRect() = 0;
    virtual ~ModernRenderer() = default;
};

// 遗留类（旧系统）
class OldPainter {
public:
    void drawRect(int r) { std::cout << "Old: rect with corner " << r << '\n'; }
};

// 适配器
class PainterAdapter : public ModernRenderer {
public:
    explicit PainterAdapter(OldPainter& op) : old_(op) {}
    void drawRoundedRect() override { old_.drawRect(5); } // 转调+翻译
private:
    OldPainter& old_;          // 组合
};

// 客户端
void clientCode(const ModernRenderer& r) { r.drawRoundedRect(); }

int main() {
    OldPainter      legacy;
    PainterAdapter  adapter(legacy);
    clientCode(adapter);      // 输出：Old: rect with corner 5
}

```

类适配器（多继承，快速但耦合高）

```cpp
class OldPainter {
public:
    void drawRect(int r) { std::cout << "Old: rect " << r << '\n'; }
};

class ModernRenderer {
public:
    virtual void drawRoundedRect() = 0;
    virtual ~ModernRenderer() = default;
};

// 同时继承接口和实现
class Adapter : public ModernRenderer, private OldPainter {
public:
    void drawRoundedRect() override { drawRect(5); } // 直接复用
};

int main() {
    Adapter a;
    a.drawRoundedRect();        // 输出：Old: rect 5
}
```

### 2.2 桥接模式（Bridge Pattern）

桥接模式的目的是将抽象与实现分离，使它们可以独立地变化，该模式通过将一个对象的抽象部分与它的实现部分分离，使它们可以独立地改变。它通过组合的方式，而不是继承的方式，将抽象和实现的部分连接起来。

实现层

```cpp
// 实现接口
class Renderer {
public:
    virtual void drawCircle(float x, float y, float r) = 0;
    virtual ~Renderer() = default;
};

// OpenGL 实现
class OpenGLRenderer : public Renderer {
public:
    void drawCircle(float x, float y, float r) override {
        std::cout << "[OpenGL] circle at (" << x << ',' << y << ") r=" << r << '\n';
    }
};

// Vulkan 实现
class VulkanRenderer : public Renderer {
public:
    void drawCircle(float x, float y, float r) override {
        std::cout << "[Vulkan] circle at (" << x << ',' << y << ") r=" << r << '\n';
    }
};
```

抽象层

```cpp
// 抽象层
class Shape {
public:
    explicit Shape(Renderer& rd) : renderer_(rd) {}
    virtual void draw() = 0;
    virtual ~Shape() = default;
protected:
    Renderer& renderer_; // 桥接指针（引用版）
};

// 具体形状：圆
class Circle : public Shape {
public:
    Circle(float x, float y, float r, Renderer& rd)
        : Shape(rd), x_(x), y_(y), r_(r) {}
    void draw() override { renderer_.drawCircle(x_, y_, r_); }
private:
    float x_, y_, r_;
};
```

客户端

```cpp
int main() {
    OpenGLRenderer gl;
    VulkanRenderer vk;

    Circle c1(10, 20, 5, gl); // OpenGL 画圆
    Circle c2(30, 40, 7, vk); // Vulkan 画圆

    c1.draw();
    c2.draw();
}
```

`Shape`中的`Render &`就是所谓的“桥”，连接抽象层和实现层。

### 2.3 过滤器模式（Filter Pattern）

过滤器模式（Filter Pattern）或标准模式（Criteria Pattern）是一种设计模式，这种模式允许开发人员使用不同的标准来过滤一组对象，通过逻辑运算以解耦的方式把它们连接起来。这种类型的设计模式属于结构型模式，它结合多个标准来获得单一标准。当需要根据多个不同的条件或标准来筛选一组对象时，过滤器模式提供了一种灵活的方式来定义这些条件，避免在客户端代码中硬编码筛选逻辑。

```cpp
// 领域对象
struct Player {
    std::string name;
    int         level;
    int         guildId;
    bool        online;
};
// 过滤器类型别名
using Filter = std::function<bool(const Player&)>;
// 原子过滤器
inline Filter minLevel(int lvl) {
    return [lvl](const Player& p) { return p.level >= lvl; };
}

inline Filter guildId(int id) {
    return [id](const Player& p) { return p.guildId == id; };
}

inline Filter isOnline() {
    return [](const Player& p) { return p.online; };
}
// 组合过滤器
inline Filter operator&&(Filter a, Filter b) {
    return [a, b](const Player& p) { return a(p) && b(p); };
}

inline Filter operator||(Filter a, Filter b) {
    return [a, b](const Player& p) { return a(p) || b(p); };
}

inline Filter operator!(Filter f) {
    return [f](const Player& p) { return !f(p); };
}
// 工具函数
std::vector<Player> filter(const std::vector<Player>& src, Filter f) {
    std::vector<Player> res;
    for (auto& p : src) if (f(p)) res.push_back(p);
    return res;
}
// 客户端代码
int main() {
    std::vector<Player> ps = {
        {"Alice", 90, 1001, true},
        {"Bob",   60, 1001, false},
        {"Carol", 80, 2002, true},
        {"Dave",  50, 1001, true}
    };

    // 1. 原子过滤
    auto highLevel = minLevel(70);
    print(filter(ps, highLevel));

    // 2. 组合过滤：公会 1001 且在线
    auto guildOnline = guildId(1001) && isOnline();
    print(filter(ps, guildOnline));

    // 3. 复杂：等级≥70 或者 (不在线且 guild=1001)
    auto complex = minLevel(70) || (!isOnline() && guildId(1001));
    print(filter(ps, complex));
}
```

### 2.4 组合模式（Composite Pattern）

组合模式（Composite Pattern），又叫部分整体模式，是用于把一组相似的对象当作一个单一的对象。组合模式依据树形结构来组合对象，用来表示部分以及整体层次。将对象组合成树形结构以表示"部分-整体"的层次结构。组合模式使得用户对单个对象和组合对象的使用具有一致性。

```cpp
#include <iostream>
#include <memory>
#include <vector>
#include <string>

// 1. Component
class Node {
public:
    virtual ~Node() = default;
    virtual void operation() const = 0;
    /* 透明模式：统一接口，Leaf 空实现 */
    virtual void add(std::shared_ptr<Node>) {}
    virtual void remove(std::shared_ptr<Node>) {}
};

// 2. Leaf 实现接口而不实现对子组件的管理
class File : public Node {
public:
    explicit File(std::string name) : name_(std::move(name)) {}
    void operation() const override {
        std::cout << "File: " << name_ << '\n';
    }
private:
    std::string name_;
};

// 3. Composite 实现接口和对子组件的管理
class Directory : public Node {
public:
    explicit Directory(std::string name) : name_(std::move(name)) {}
    void operation() const override {
        std::cout << "Directory [" << name_ << "] contains:\n";
        for (auto& c : children_) c->operation();
    }
    void add(std::shared_ptr<Node> node) override {
        children_.push_back(std::move(node));
    }
    void remove(std::shared_ptr<Node> node) override {
        children_.erase(
            std::remove(children_.begin(), children_.end(), node),
            children_.end());
    }
private:
    std::string name_;
    std::vector<std::shared_ptr<Node>> children_;
};

// 4. 客户端
int main() {
    auto root = std::make_shared<Directory>("root");
    root->add(std::make_shared<File>("readme.txt"));
  
    auto src = std::make_shared<Directory>("src");
    src->add(std::make_shared<File>("main.cpp"));
    src->add(std::make_shared<File>("utils.h"));
    //src是复合节点，root也是复合节点，root包含src
    //root中有一个src节点，operation时从root遍历到src，在从src遍历其子节点
    root->add(src);
    root->operation();          // 统一接口，递归遍历
    return 0;
}
```

### 2.5 装饰器模式（Decorator Pattern）

装饰器模式（Decorator Pattern）允许向一个现有的对象添加新的功能，同时又不改变其结构。这种类型的设计模式属于结构型模式，它是作为现有的类的一个包装。

装饰器模式通过将对象包装在装饰器类中，以便动态地修改其行为。

装饰器 = 继承接口 + 组合实现。

这种模式创建了一个装饰类，用来包装原有的类，并在保持类方法签名完整性的前提下，提供了额外的功能。

~~~cpp
// 目标接口
// 饮料类
class Beverage {
public:
    virtual ~Beverage() = default;
    virtual std::string description() const = 0;
    virtual double cost() const = 0;
};
// 具体组件
// 浓缩咖啡类
class Espresso : public Beverage {
public:
    std::string description() const override { return "Espresso"; }
    double cost() const override { return 1.99; }
};
// 装饰器基类 Decorator 包装目标基类
class Decorator : public Beverage {
protected:
    std::unique_ptr<Beverage> wrapped_; // 组合代替继承
public:
    explicit Decorator(std::unique_ptr<Beverage> b)
        : wrapped_(std::move(b)) {}
};
// 具体装饰器 实现新的组件\装饰器 可以包装自身或具体组件
class Milk : public Decorator {
public:
    using Decorator::Decorator;
    std::string description() const override {
        return wrapped_->description() + ", Milk";
    }
    double cost() const override { return wrapped_->cost() + 0.30; }
};

class Sugar : public Decorator {
public:
    using Decorator::Decorator;
    std::string description() const override {
        return wrapped_->description() + ", Sugar";
    }
    double cost() const override { return wrapped_->cost() + 0.20; }
};
// 客户端
int main() {
    // 一杯 Espresso + Milk + Sugar
    auto coffee = std::make_unique<Sugar>(
                    std::make_unique<Milk>(
                        std::make_unique<Espresso>()));

    std::cout << coffee->description() << "  $" << coffee->cost() << '\n';
    // Espresso, Milk, Sugar  $2.49
    return 0;
}
~~~

### 2.6 外观模式（Facade Pattern）

外观模式（Facade Pattern）隐藏系统的复杂性，并向客户端提供了一个客户端可以访问系统的接口。为一个复杂的子系统提供一个一致的高层接口。这样，客户端代码就可以通过这个简化的接口与子系统交互，而不需要了解子系统内部的复杂性。

```cpp
// ---------- 子系统（无需修改）----------
class Audio {
public:
    void init()   { std::cout << "Audio init\n"; }
    void play()   { std::cout << "Audio play\n"; }
    void stop()   { std::cout << "Audio stop\n"; }
};

class Video {
public:
    void init()   { std::cout << "Video init\n"; }
    void render() { std::cout << "Video render 4K\n"; }
    void stop()   { std::cout << "Video stop\n"; }
};

class Network {
public:
    void connect() { std::cout << "Network connect CDN\n"; }
    void disconnect() { std::cout << "Network disconnect\n"; }
};

class Light {
public:
    void dim(int level) { std::cout << "Light dim to " << level << "%\n"; }
    void restore()      { std::cout << "Light restore\n"; }
};

// ---------- 外观类（唯一入口） ----------
class MediaFacade {
public:
    MediaFacade()
        : audio_(std::make_unique<Audio>()),
          video_(std::make_unique<Video>()),
          net_(std::make_unique<Network>()),
          light_(std::make_unique<Light>()) {}

    /* 一键观影 */
    void watchMovie() {
        light_->dim(30);
        audio_->init();
        video_->init();
        net_->connect();
        audio_->play();
        video_->render();
    }

    /* 一键结束 */
    void endMovie() {
        audio_->stop();
        video_->stop();
        net_->disconnect();
        light_->restore();
    }

private:
    std::unique_ptr<Audio>   audio_;
    std::unique_ptr<Video>   video_;
    std::unique_ptr<Network> net_;
    std::unique_ptr<Light>   light_;
};

// ---------- 客户端 ----------
int main() {
    MediaFacade media;
    media.watchMovie();
    std::cout << "--- enjoy film ---\n";
    media.endMovie();
    return 0;
}
```

### 2.7 享元模式（Flyweight Pattern）

享元模式（Flyweight Pattern）主要用于减少创建对象的数量，以减少内存占用和提高性能。这种类型的设计模式属于结构型模式，它提供了减少对象数量从而改善应用所需的对象结构的方式。

享元模式尝试重用现有的同类对象，如果未找到匹配的对象，则创建新对象。

```cpp
// 固有数据（享元，只读共享）
struct SoldierType {
    std::string model;   // 网格文件
    std::string texture; // 贴图文件
    int         hpMax;   // 最大血量
    // 注意：这里全是**不会变**的共享数据
};
// 享元工厂（负责创建 / 复用）
class SoldierTypeFactory {
public:
    const SoldierType* get(const std::string& key) {
        auto it = pool_.find(key);
        if (it != pool_.end()) return &it->second;
        // 第一次用 → 构造并插入
        return &pool_.emplace(key, SoldierType{key + ".mesh", key + ".tex", 100}).first->second;
    }
    std::size_t size() const { return pool_.size(); } // 调试用
private:
    std::unordered_map<std::string, SoldierType> pool_; // 无指针，直接存对象
};
// 可变数据（外部状态，Extrinsic）
struct Soldier {
    const SoldierType* type; // 指向享元（固有）
    float              x, z;   // 世界坐标（可变）
    int                hp;     // 当前血量（可变）
};
// 客户端
int main() {
    SoldierTypeFactory factory;
    std::vector<Soldier> army;
    army.reserve(100'000);

    // 两种兵模
    const SoldierType* archer = factory.get("Archer");
    const SoldierType* knight = factory.get("Knight");

    for (std::size_t i = 0; i < 50'000; ++i) {
        // army.push_back({SoldiderType , pos.x , pos.z , hp});
        army.push_back({archer, std::rand() % 1000 * 1.0f, std::rand() % 1000 * 1.0f, 100});
        army.push_back({knight, std::rand() % 1000 * 1.0f, std::rand() % 1000 * 1.0f, 100});
    }

    std::cout << "Total soldiers: " << army.size() << '\n';
    std::cout << "Total SoldierType objects: " << factory.size() << '\n'; // 仅 2 份
}

```

### 2.8 代理模式（Proxy Pattern）

在代理模式（Proxy Pattern）中，一个类代表另一个类的功能，这种类型的设计模式属于结构型模式。

代理模式通过引入一个代理对象来控制对原对象的访问。代理对象在客户端和目标对象之间充当中介，负责将客户端的请求转发给目标对象，同时可以在转发请求前后进行“额外的处理”。

~~~cpp
// 目标接口（抽象主题）
class Image {
public:
    virtual ~Image() = default;
    virtual void display() = 0;
};
// 真实主题（RealImage）
class RealImage : public Image {
public:
    explicit RealImage(std::string file) : fileName_(std::move(file)) {
        loadFromDisk(); // 构造函数里真加载
    }
    void display() override {
        std::cout << "[Real] Displaying " << fileName_ << '\n';
    }
private:
    void loadFromDisk() {
        std::cout << "[Real] Loading heavy file: " << fileName_ << '\n';
        // 模拟耗时
    }
    std::string fileName_;
};
//  代理主题（ProxyImage）
class ProxyImage : public Image {
public:
    explicit ProxyImage(std::string file)
        : fileName_(std::move(file)), real_(nullptr) {}

    void display() override {
        if (!real_) { // 第一次才创建真对象
            real_ = std::make_unique<RealImage>(fileName_);
            cacheToDisk(); // 可选附加逻辑
        }
        real_->display();
    }
private:
    void cacheToDisk() { std::cout << "[Proxy] Cached thumbnail on disk\n"; }

    std::string fileName_;
    std::unique_ptr<RealImage> real_; // 懒加载句柄
};
// 客户端
int main() {
    Image* img = new ProxyImage("4K_wallpaper.jpg"); // 代理替身
    std::cout << "---- first call ----\n";
    img->display(); // 触发加载 + 缓存
    std::cout << "---- second call ----\n";
    img->display(); // 直接复用，无加载
    delete img;
}
~~~

`Proxy`内部持有`Real`，并在第一次`disaplay（）`时创建。

## 3. ⅩⅡ个行为型模式

### 3.1 责任链模式（Chain of Responsibility Pattern）

责任链模式（Chain of Responsibility Pattern）为请求创建了一个接收者对象的链。这种模式给予请求的类型，对请求的发送者和接收者进行解耦。

责任链模式通过将多个处理器（处理对象）以链式结构连接起来，使得请求沿着这条链传递，直到有一个处理器处理该请求为止。

责任链模式允许多个对象都有机会处理请求，从而避免请求的发送者和接收者之间的耦合关系。将这些对象连成一条链，并沿着这条链传递请求。

```cpp
//请求结构
struct LeaveRequest {
    std::string name;
    int         days;
};
// 抽象处理者
class Handler {
public:
    explicit Handler(std::shared_ptr<Handler> next = nullptr)
        : next_(std::move(next)) {}

    /* 入口：沿链传递 */
    void handle(const LeaveRequest& req) {
        if (canHandle(req))                     // 自己能处理？
            process(req);
        else if (next_)                         // 交给后继
            next_->handle(req);
        else
            std::cout << "Chain end: no handler for " << req.name << '\n';
    }

protected:
    /* 子类只需实现这两个 */
    virtual bool canHandle(const LeaveRequest& req) const = 0;
    virtual void process(const LeaveRequest& req) const = 0;

private:
    std::shared_ptr<Handler> next_; // 后继链
};
// 具体处理者
class TeamLeader : public Handler {
public:
    using Handler::Handler;
protected:
    bool canHandle(const LeaveRequest& req) const override {
        return req.days <= 2;
    }
    void process(const LeaveRequest& req) const override {
        std::cout << "[TeamLeader] approved " << req.name
                  << " " << req.days << " days\n";
    }
};

class Manager : public Handler {
protected:
    bool canHandle(const LeaveRequest& req) const override {
        return req.days <= 5;
    }
    void process(const LeaveRequest& req) const override {
        std::cout << "[Manager] approved " << req.name
                  << " " << req.days << " days\n";
    }
};

class HR : public Handler {
protected:
    bool canHandle(const LeaveRequest& req) const override {
        return req.days <= 30;
    }
    void process(const LeaveRequest& req) const override {
        std::cout << "[HR] approved " << req.name
                  << " " << req.days << " days\n";
    }
};

class AutoReject : public Handler { // 链尾兜底
protected:
    bool canHandle(const LeaveRequest&) const override { return true; }
    void process(const LeaveRequest& req) const override {
        std::cout << "[AutoReject] " << req.days
                  << " days too much for " << req.name << '\n';
    }
};
// 客户端
int main() {
    /* 手动组装责任链：TeamLeader → Manager → HR → AutoReject */
    auto chain = std::make_shared<TeamLeader>(
                   std::make_shared<Manager>(
                     std::make_shared<HR>(
                       std::make_shared<AutoReject>())));

    chain->handle({"Alice", 1});   // TeamLeader 处理
    chain->handle({"Bob", 4});     // Manager 处理
    chain->handle({"Carol", 20});  // HR 处理
    chain->handle({"Dave", 50});   // AutoReject 兜底
}
```

### 3.2 命令模式（Command Pattern）

命令模式（Command Pattern）是一种数据驱动的设计模式，它属于行为型模式。

命令模式将一个请求封装为一个对象，从而使你可以用不同的请求对客户进行参数化，对请求排队或记录请求日志，以及支持可撤销的操作。

```cpp
// 抽象命令
class Command {
public:
    virtual ~Command() = default;
    virtual void execute() = 0;
    virtual void undo()    = 0;
};
// 接收者
class Canvas {
public:
    void drawCircle(float x, float y, float r) {
        circles_.emplace_back(x, y, r);
        std::cout << "[Canvas] draw circle at (" << x << ',' << y << ") r=" << r << '\n';
    }
    void moveCircle(std::size_t idx, float dx, float dy) {
        auto& c = circles_[idx];
        c.x += dx; c.y += dy;
        std::cout << "[Canvas] move circle #" << idx << " by (" << dx << ',' << dy << ")\n";
    }
    const auto& circles() const { return circles_; }
private:
    struct Circle { float x, y, r; };
    std::vector<Circle> circles_;
};
// 具体命令
class DrawCircleCmd : public Command {
public:
    DrawCircleCmd(Canvas& c, float x, float y, float r)
        : canvas_(c), x_(x), y_(y), r_(r), idx_(0) {}
    void execute() override {
        canvas_.drawCircle(x_, y_, r_);
        idx_ = canvas_.circles().size() - 1; // 记录索引供撤销
    }
    void undo() override {
        canvas_.circles().pop_back(); // 简单移除最后一个
        std::cout << "[Undo] remove last circle\n";
    }
private:
    Canvas& canvas_;
    float x_, y_, r_;
    std::size_t idx_;
};

class MoveCircleCmd : public Command {
public:
    MoveCircleCmd(Canvas& c, std::size_t idx, float dx, float dy)
        : canvas_(c), idx_(idx), dx_(dx), dy_(dy) {}
    void execute() override { canvas_.moveCircle(idx_, dx_, dy_); }
    void undo()    override { canvas_.moveCircle(idx_, -dx_, -dy_); }
private:
    Canvas& canvas_;
    std::size_t idx_;
    float dx_, dy_;
};
// 宏命令
class MacroCommand : public Command {
public:
    void add(std::shared_ptr<Command> cmd) { cmds_.push_back(std::move(cmd)); }
    void execute() override {
        for (auto& c : cmds_) c->execute();
    }
    void undo() override {
        for (auto it = cmds_.rbegin(); it != cmds_.rend(); ++it)
            (*it)->undo(); // 逆序撤销
    }
private:
    std::vector<std::shared_ptr<Command>> cmds_;
};
// 调用者
class CommandInvoker {
public:
    void execute(std::shared_ptr<Command> cmd) {
        cmd->execute();
        undo_.push(cmd);
        // 执行新命令后清空重做栈
        while (!redo_.empty()) redo_.pop();
    }
    void undo() {
        if (undo_.empty()) return;
        auto cmd = undo_.top();
        undo_.pop();
        cmd->undo();
        redo_.push(cmd);
    }
    void redo() {
        if (redo_.empty()) return;
        auto cmd = redo_.top();
        redo_.pop();
        cmd->execute();
        undo_.push(cmd);
    }
private:
    std::stack<std::shared_ptr<Command>> undo_;
    std::stack<std::shared_ptr<Command>> redo_;
};
// 客户端
int main() {
    Canvas canvas;
    CommandInvoker invoker;

    invoker.execute(std::make_shared<DrawCircleCmd>(canvas, 10, 10, 5));
    invoker.execute(std::make_shared<DrawCircleCmd>(canvas, 20, 20, 3));

    auto macro = std::make_shared<MacroCommand>();
    macro->add(std::make_shared<MoveCircleCmd>(canvas, 0, 5, 0));
    macro->add(std::make_shared<MoveCircleCmd>(canvas, 1, 0, 5));
    invoker.execute(macro);

    std::cout << "---- undo 2 steps ----\n";
    invoker.undo(); // 撤销宏
    invoker.undo(); // 撤销第二个圆

    std::cout << "---- redo 1 step ----\n";
    invoker.redo(); // 重做第二个圆
}

```

### 3.3 解释器模式（Interpreter Pattern）

解释器模式（Interpreter Pattern）提供了评估语言的语法或表达式的方式，它属于行为型模式。

解释器模式给定一个语言，定义它的文法的一种表示，并定义一个解释器，这个解释器使用该表示来解释语言中的句子。

当某一特定类型的问题频繁出现，并且可以通过一种简单的语言来表达这些问题的实例时，解释器模式用于构建一个能够解释特定语言或文法的句子的解释器。

个人理解，解释器模式就是将给定的语言string/自定义语言解释为符合程序的表达式/语法，类似编译器将代码编译为机器语言，主要在自定义语法时使用。

```cpp
// 抽象表达式（Abstract Expression）：定义了解释器的抽象接口，声明了解释操作的方法，通常是一个抽象类或接口。
class IExpression{
    public:
        virtual bool interpret(const std::string& context) = 0;
        virtual ~IExpression() = default;
};
// 终结符表达式（Terminal Expression）：实现了抽象表达式接口，表示语言中的基本符号或原子表达式。
class TerminalExpression : public IExpression{
    private:
        std::string data;
    public:
        TerminalExpression(const std::string& str):data(str){}
        bool interpret(const std::string& context) override {
            return context.find(data) != std::string::npos;
        }
};
// 非终结符表达式（Non-terminal Expression）：实现了抽象表达式接口，表示语言中的复合结构，通常包含一个或多个子表达式。
class OrExpression : public IExpression{
    private:
        IExpression* expr1;
        IExpression* expr2;
    public:
        OrExpression(IExpression* exp1, IExpression* exp2):expr1(exp1), expr2(exp2){}
        bool interpret(const std::string& context) override {
            return expr1->interpret(context) || expr2->interpret(context);
        }
        ~OrExpression() {
            delete expr1;
            delete expr2;
        }
};
class AndExpression : public IExpression{
    private:
        IExpression* expr1;
        IExpression* expr2;
    public:
        AndExpression(IExpression* exp1, IExpression* exp2):expr1(exp1), expr2(exp2){}
        bool interpret(const std::string& context) override {
            return expr1->interpret(context) && expr2->interpret(context);
        }
        ~AndExpression() {
            delete expr1;
            delete expr2;
        }
};
// 客户端（Client）：创建并配置具体的解释器对象，并将需要解释的表达式传递给解释器进行解释。
// robert 和 john 中有一个男人 
static IExpression * GetMaleExpression ()
{
    IExpression * robert = new TerminalExpression("Robert");
    IExpression * john = new TerminalExpression("John");
    return new OrExpression(robert,john);
}
// jule是人妻
static IExpression * GetMarriedFemaleExpression()
{
    IExpression * jule = new TerminalExpression("Jule");
    IExpression * married = new TerminalExpression("Married");
    return new AndExpression(jule,married);
}
int main()
{
    IExpression * isMale = GetMaleExpression();
    IExpression * isMarriedFemale = GetMarriedFemaleExpression(); 

    std::string ans1 =  isMale->interpret("John") ? "yes": "no";
    std::string ans2 =  isMarriedFemale->interpret("Married Jule") ? "yes": "no";
  
    std::cout<< "one of john and robert is male? " + ans1 <<std::endl;
    std::cout<< "jule is married female? " + ans2 <<std::endl;

    return 0;  
}
```

### 3.4 迭代器模式（Iterator Pattern）

迭代器模式提供一种方法顺序访问一个聚合对象中的各个元素，而又不暴露其内部的表示。

在`C++`中，容器类`vector`、`list`本身提供迭代器。

```cpp
// 迭代器接口（Iterator）：定义了访问和遍历聚合对象中各个元素的方法，
// 通常包括获取下一个元素、判断是否还有元素、获取当前位置等方法。
template <typename T>
class Iterator{
public:
    virtual bool hasNext() = 0;
    virtual T   next() = 0;
    virtual ~Iterator() =default;
};
// 聚合对象接口（Aggregate）：定义了创建迭代器对象的接口，通常包括一个工厂方法用于创建迭代器对象。
template <typename T>
class IContainer{
public:
    virtual Iterator<T> * getIterator() = 0;
    virtual ~IContainer() = default;
};
// 具体聚合对象（Concrete Aggregate）：实现了聚合对象接口,
// 负责创建具体的迭代器对象，并提供需要遍历的数据。
template <typename T>
class MyContainer : public IContainer<T>{
private:
    T * arr;
    int size;
// 具体迭代器（Concrete Iterator）：实现了迭代器接口，负责对聚合对象进行遍历和访问，同时记录遍历的当前位置。
    class MyIterator : public Iterator<T>{
    private:
        int index;
        T * _arr;
        int _size;
    public:
        MyIterator(T * a , int s) :index(0),_arr(a),_size(s) {}
        bool hasNext() override{
            return index < _size;
        }
        T  next() override{
            if(hasNext())
                return  (_arr[index++]);
            throw std::out_of_range("Out of Range");
        }
    };
public:
    MyContainer(T * a, int s) : arr(a), size(s) {}

    Iterator<T> * getIterator() override{
        return new MyIterator(arr , size);
    }
};
// 客户端
int main() {
    int arr[] = {1,2,3,4,5};
    MyContainer<int> container(arr, 5);
    Iterator<int> * it = container.getIterator();
    while(it->hasNext()){
        std::cout << it->next() << std::endl;
    }
    delete it;
    return 0;
}
```

### 3.5 中介者模式（Mediator Pattern）

中介者模式（Mediator Pattern）定义了一个中介对象来封装一系列对象之间的交互。中介者使各对象之间不需要显式地相互引用，从而使其耦合松散，且可以独立地改变它们之间的交互。

```cpp
// 前向声明
class User;  
// 具体中介者（Concrete Mediator）：实现了中介者接口，负责实现各个同事对象之间的通信逻辑。
// 它会维护一个对各个同事对象的引用，并协调它们的交互。
class Chatroom {
public:
    void broadcast(const User& sender, const std::string& msg);
    void join(std::shared_ptr<User> user);
private:
    std::vector<std::weak_ptr<User>> users;
};
// 具体同事对象（Concrete Colleague）：实现了同事对象接口，是真正参与到交互中的对象。
// 它会将自己的消息发送给中介者，由中介者转发给其他同事对象。
class User {
public:
    explicit User(std::string _name) : name(std::move(_name)), room(nullptr) {}
    void send(const std::string& msg);
    void receive(const std::string& sender, const std::string& msg);
    std::string name;
    Chatroom* room = nullptr;   
};

// ========== 实现 ==========
void Chatroom::broadcast(const User& sender, const std::string& msg) {
    std::cout << "[Room] " << sender.name << msg << '\n';
    for (auto& wk : users) {
        if (auto u = wk.lock())
            if (u.get() != &sender)
                u->receive(sender.name, msg);
    }
}
void Chatroom::join(std::shared_ptr<User> user) {
    users.push_back(user);
    user->room = this;
    broadcast(*user, " joined.");
}
void User::send(const std::string& msg) {
    if (room) room->broadcast(*this, ": " + msg);
}
void User::receive(const std::string& sender, const std::string& msg) {
    std::cout << name << " received from " << sender << msg << '\n';
}

// ========== main ==========
int main() {
    auto room = std::make_shared<Chatroom>();   
    auto alice = std::make_shared<User>("Alice");
    auto bob   = std::make_shared<User>("Bob");
    room->join(alice);
    room->join(bob);
    alice->send("Hello guys!");
    bob->send("Hi Alice");
    return 0;  
}
```

### 3.6 备忘录模式（Memento Pattern）

备忘录模式（Memento Pattern）保存一个对象的某个状态，以便在适当的时候恢复对象，备忘录模式属于行为型模式。

备忘录模式允许在不破坏封装性的前提下，捕获和恢复对象的内部状态。

备忘录模式包含以下几个主要角色：

* **备忘录（Memento）**：负责存储原发器对象的内部状态。备忘录可以保持原发器的状态的一部分或全部信息。
* **原发器（Originator）**：创建一个备忘录对象，并且可以使用备忘录对象恢复自身的内部状态。原发器通常会在需要保存状态的时候创建备忘录对象，并在需要恢复状态的时候使用备忘录对象。
* **负责人（Caretaker）**：负责保存备忘录对象，但是不对备忘录对象进行操作或检查。负责人只能将备忘录传递给其他对象。

```cpp
#include <iostream>
#include <list>
// 备忘录（Memento）：负责存储原发器对象的内部状态。备忘录可以保持原发器的状态的一部分或全部信息。
class Memento{
private:
    std::string state;
public:
    Memento(const std::string& state): state(state) {}
    std::string getState() const { return state; }
};
// 原发器（Originator）：创建一个备忘录对象，并且可以使用备忘录对象恢复自身的内部状态。
// 原发器通常会在需要保存状态的时候创建备忘录对象，并在需要恢复状态的时候使用备忘录对象。
class Originator{
private:
    std::string state;
public:
    void setState(const std::string& state) { this->state = state; }
    std::string getState() const { return state; }
    Memento * createMemento() { return new Memento(state); }
    void restoreMemento(Memento* memento) { state = memento->getState(); }
};
// 负责人（Caretaker）：负责保存备忘录对象，但是不对备忘录对象进行操作或检查。
// 负责人只能将备忘录传递给其他对象。
class CareTaker{
private:
    std::list<Memento *> mementos;
public:
    void add(Memento * state){ mementos.push_back(state); } 
    Memento * get(int index){
        auto it = mementos.begin();
        std::advance(it, index);
        return *it;
    }
};
int main() {
    Originator originator;
    CareTaker careTaker;
  
    originator.setState("State #1");
    // 原发器将状态2存储进备忘录管理表careTaker
    originator.setState("State #2");
    careTaker.add( originator.createMemento());
    // 原发器将状态3存储进备忘录管理表careTaker
    originator.setState("State #3");
    careTaker.add( originator.createMemento());
    originator.setState("State #4");

    std::cout<<"Current State: " + originator.getState()    <<std::endl;
    // 原发器从备忘录管理表careTaker读取第一次存储
    originator.restoreMemento(careTaker.get(0));
    std::cout<<"First saved State: " + originator.getState()<<std::endl;
    // 原发器从备忘录管理表careTaker读取第二次存储
    originator.restoreMemento(careTaker.get(1));
    std::cout<<"Second saved State: " + originator.getState()<<std::endl;;
}
```

### 3.7 观察者模式（Observe Pattern）

观察者模式是一种行为型设计模式，它定义了一种一对多的依赖关系，当一个对象的状态发生改变时，其所有依赖者都会收到通知并自动更新。

当对象间存在一对多关系时，则使用观察者模式（Observer Pattern）。比如，当一个对象被修改时，则会自动通知依赖它的对象。观察者模式属于行为型模式。

**观察者模式包含以下几个核心角色：**

* 主题（Subject）：也称为被观察者或可观察者，它是具有状态的对象，并维护着一个观察者列表。主题提供了添加、删除和通知观察者的方法。
* 观察者（Observer）：观察者是接收主题通知的对象。观察者需要实现一个更新方法，当收到主题的通知时，调用该方法进行更新操作。
* 具体主题（Concrete Subject）：具体主题是主题的具体实现类。它维护着观察者列表，并在状态发生改变时通知观察者。
* 具体观察者（Concrete Observer）：具体观察者是观察者的具体实现类。它实现了更新方法，定义了在收到主题通知时需要执行的具体操作。

```cpp
#include <iostream>
#include <list>
#include <memory>
#include <functional>
#include <algorithm>
// 具体观察者（Concrete Observer）：具体观察者是观察者的具体实现类。
// 它实现了更新方法，定义了在收到主题通知时需要执行的具体操作。
class Observer{
    public:
    explicit  Observer(std::string name = "Default"):name(name){}
    void update() const{
        std::cout << name +" Observer  updated" << std::endl;
    }
    private:
        std::string name;
};
// 具体主题（Concrete Subject）：具体主题是主题的具体实现类。它维护着观察者列表，
// 并在状态发生改变时通知观察者。
class Subject{
    private:
        using observerFunction = std::function<void()>;
        std::list<observerFunction> observers;
    public:
        using Token = std::shared_ptr<bool>;
        Token attach(const observerFunction & observer){
            auto token = std::make_shared<bool>(true);
            observers.emplace_back([token , observer]() {if(* token) observer();});
            return token;
        }
        void detach(Token token){
            *token = false;            // 标记失效
            // 可选：惰性擦除所有失效项
            observers.erase(std::remove_if(observers.begin(), observers.end(),
                          [](const auto& fn) { return !fn; }),
            observers.end());  
        }
        void notify(){
            for(auto & observer : observers){
                observer();
            }
        }
};
int main()
{
  
    Subject subject;
    Observer observer1("A");
    Observer observer2("B");

    auto token1 = subject.attach([&]() { observer1.update(); });
    auto token2 = subject.attach([&]() { observer2.update(); });

    subject.notify();

    subject.detach(std::move(token1));
    subject.detach(std::move(token2));
  
    return 0;
}   
```

### 3.8 状态模式（State Pattern）

在状态模式中，我们创建表示各种状态的对象和一个行为随着状态对象改变而改变的 context 对象。

状态模式允许对象在内部状态改变时改变其行为，使得对象在不同的状态下有不同的行为表现。通过将每个状态封装成独立的类，可以避免使用大量的条件语句来实现状态切换。

状态模式包含以下几个主要角色：

* **上下文（Context）**：定义了客户感兴趣的接口，并维护一个当前状态对象的引用。上下文可以通过状态对象来委托处理状态相关的行为。
* **状态（State）**：定义了一个接口，用于封装与上下文相关的一个状态的行为。
* **具体状态（Concrete State）**：实现了状态接口，负责处理与该状态相关的行为。具体状态对象通常会在内部维护一个对上下文对象的引用，以便根据不同的条件切换到不同的状态。

```cpp
#include <iostream>

// 1. 前向声明所有具体状态
class IdleState;
class RunState;
class JumpState;
class Player;

// 2. 状态接口
class IState {
public:
    virtual ~IState() = default;
    virtual void enter(Player& p)  {}
    virtual void update(Player& p) = 0;
    virtual void exit(Player& p)   {}
    virtual const char* name() const = 0;
};

// 3. Player 声明（不实现）
class Player {
public:
    Player();                          // 构造函数定义放后面
    void update();
    void handleInput(char c);
    void changeState(IState* newState); // 定义放后面
private:
    IState* current = nullptr;
};

// 4. 具体状态类（可安全调用 Player::changeState 的声明）
class IdleState : public IState {
public:
    static IState* instance() { static IdleState ins; return &ins; }
    void update(Player& p) override;
    const char* name() const override { return "Idle"; }
private:
    IdleState() = default;
};

class RunState : public IState {
public:
    static IState* instance() { static RunState ins; return &ins; }
    void update(Player& p) override;
    const char* name() const override { return "Run"; }
private:
    RunState() = default;
};

class JumpState : public IState {
public:
    static IState* instance() { static JumpState ins; return &ins; }
    void update(Player& p) override;
    const char* name() const override { return "Jump"; }
private:
    JumpState() = default;
};

// 5. 实现 Player 的成员函数（所有 instance() 已可见）
inline Player::Player() { changeState(IdleState::instance()); }

inline void Player::update() {
    if (current) current->update(*this);
}

inline void Player::handleInput(char c) {
    switch (c) {
    case 'r': changeState(RunState::instance()); break;
    case 'j': changeState(JumpState::instance()); break;
    case 'i': changeState(IdleState::instance()); break;
    }
}

inline void Player::changeState(IState* newState) {
    if (current) current->exit(*this);
    current = newState;
    if (current) current->enter(*this);
}

// 6. 状态实现（可调用 Player 完整接口）
inline void IdleState::update(Player& p) {
    std::cout << "[Idle] 站立\n";
}
inline void RunState::update(Player& p) {
    std::cout << "[Run] 奔跑\n";
}
inline void JumpState::update(Player& p) {
    std::cout << "[Jump] 跳跃\n";
    static int tick = 0;   // 单例，全局仅一份
    if (++tick > 2) {
        tick = 0;
        p.changeState(IdleState::instance());
    }
}

// 7. 客户端
int main() {
    Player player;
    player.handleInput('r');
    player.update();
    player.handleInput('j');
    for (int i = 0; i < 3; ++i) player.update();
    player.handleInput('i');
    player.update();
    return 0;
}
```

### 3.9 空对象模式（Null Object Pattern）

在空对象模式（Null Object Pattern）中，一个空对象取代 NULL 对象实例的检查。Null 对象不是检查空值，而是反应一个不做任何动作的关系。这样的 Null 对象也可以在数据不可用的时候提供默认的行为。

在空对象模式中，我们创建一个指定各种要执行的操作的抽象类和扩展该类的实体类，还创建一个未对该类做任何实现的空对象类，该空对象类将无缝地使用在需要检查空值的地方。

空对象模式包含以下几个主要角色：

* **抽象对象（Abstract Object）**：定义了客户端所期望的接口。这个接口可以是一个抽象类或接口。
* **具体对象（Concrete Object）**：实现了抽象对象接口的具体类。这些类提供了真实的行为。
* **空对象（Null Object）**：实现了抽象对象接口的空对象类。这个类提供了默认的无效行为，以便在对象不可用或不可用时使用。它可以作为具体对象的替代者，在客户端代码中代替空值检查。

```cpp
#include <iostream>
#include <fstream>
#include <memory>
#include <mutex>

class ILogger {
public:
    virtual ~ILogger() = default;
    virtual void log(const std::string& msg) = 0;
    virtual const std::string name() const = 0;
};

// 1. 空对象：禁止拷贝/删除，避免误用
class NullLogger final : public ILogger {
public:
    static ILogger* instance() { static NullLogger ins; return &ins; }
    void log(const std::string&) override {  }
    const std::string name() const override { return "NullLogger"; }
private:
    NullLogger() = default;
    ~NullLogger()  = default;
    NullLogger(const NullLogger&) = delete;
    void operator=(const NullLogger&) = delete;
};

// 2. 文件日志器：打开失败抛异常
class FileLogger final : public ILogger {
public:
    explicit FileLogger(const std::string path) : ofs(path, std::ios::app) {
        if (!ofs.is_open())
            throw std::runtime_error("FileLogger: cannot open file");
    }
    void log(const std::string& msg) override {
        std::lock_guard<std::mutex> lk(mtx);
        ofs << msg << '\n';
    }
    const std::string name() const override { return "FileLogger"; }
private:
    std::ofstream ofs;
    std::mutex mtx;
};

// 3. 日志器工厂：返回单例
ILogger* createLogger(bool enable) {
    if (enable) return new FileLogger("app.log");
    return NullLogger::instance();   
}
int main() {

    auto logger = createLogger(false);
    logger->log("hello world");
    std::cout << "before main return: " << logger->name() << '\n';
    return 0;
}
```

### 3.10 策略模式（Strategy Pattern）

在策略模式定义了一系列算法或策略，并将每个算法封装在独立的类中，使得它们可以互相替换。通过使用策略模式，可以在运行时根据需要选择不同的算法，而不需要修改客户端代码。

**策略模式包含以下几个核心角色：**

* 环境（Context）：维护一个对策略对象的引用，负责将客户端请求委派给具体的策略对象执行。环境类可以通过依赖注入、简单工厂等方式来获取具体策略对象。
* 抽象策略（Abstract Strategy）：定义了策略对象的公共接口或抽象类，规定了具体策略类必须实现的方法。
* 具体策略（Concrete Strategy）：实现了抽象策略定义的接口或抽象类，包含了具体的算法实现。

```cpp
#include <vector>
#include <iostream>
#include <algorithm>

class ISortStrategy {
public:
    virtual ~ISortStrategy() = default;
    virtual void sort(std::vector<int>& data) = 0;
    virtual const char* name() const = 0;
};

// 上下文：持有策略接口
class Sorter {
public:
    explicit Sorter(ISortStrategy * s) : strategy(std::move(s)) {}
    // 运行时换策略
    void setStrategy(ISortStrategy * s) {
        strategy = std::move(s);
    }
    void sort(std::vector<int>& data) {
        strategy->sort(data);
    }
    const char* strategyName() const {
        return strategy->name();
    }
private:
    ISortStrategy * strategy;
};

class BubbleSort final : public ISortStrategy {
public:
    static ISortStrategy* instance() { static BubbleSort ins; return &ins; }
    void sort(std::vector<int>& data) override {
        for (size_t i = 0; i < data.size(); ++i)
            for (size_t j = 0; j < data.size() - i - 1; ++j)
                if (data[j] > data[j + 1]) std::swap(data[j], data[j + 1]);
    }
    const char* name() const override { return "BubbleSort"; }
private:
    BubbleSort() = default;
};

class QuickSort final : public ISortStrategy {
public:
    static ISortStrategy * instance() {
        static QuickSort ins;
        return &ins;
    }
    void sort(std::vector<int>& data) override {
        quickSort(data, 0, data.size() - 1);
    }
    const char* name() const override { return "QuickSort"; }
private:
    void quickSort(std::vector<int>& v, int l, int r) {
        if (l >= r) return;
        int pivot = v[l];
        int i = l, j = r;
        while (i < j) {
            while (i < j && v[j] >= pivot) --j;
            if (i < j) v[i++] = v[j];
            while (i < j && v[i] <= pivot) ++i;
            if (i < j) v[j--] = v[i];
        }
        v[i] = pivot;
        quickSort(v, l, i - 1);
        quickSort(v, i + 1, r);
    }
    QuickSort() = default;
};

class StdSortStrategy final : public ISortStrategy {
public:
    static ISortStrategy* instance() { static StdSortStrategy ins; return &ins; }
    void sort(std::vector<int>& data) override {
        std::sort(data.begin(), data.end());
    }
    const char* name() const override { return "StdSort"; }
private:
    StdSortStrategy() = default;
};

int main() {
    std::vector<int> data{9, 3, 7, 1, 6};

    // 1. 传统策略切换
    Sorter sorter(BubbleSort::instance());
    sorter.sort(data);
    std::cout << "After " << sorter.strategyName() << ": ";
    for (int v : data) std::cout << v << ' ';
    std::cout << '\n';

    // 2. 运行时换策略
    sorter.setStrategy(QuickSort::instance());
    sorter.sort(data);
    std::cout << "After " << sorter.strategyName() << ": ";
    for (int v : data) std::cout << v << ' ';
    std::cout << '\n';

    // 3. 运行时换策略
    sorter.setStrategy(StdSortStrategy::instance());
    sorter.sort(data);
    std::cout << "After " << sorter.strategyName() << ": ";
    for (int v : data) std::cout << v << ' ';
    std::cout << '\n';
    return 0;
}
```

### 3.11 模版模式（Template Pattern）

在模板模式（Template Pattern）中，一个抽象类公开定义了执行它的方法的方式/模板。它的子类可以按需要重写方法实现，但调用将以抽象类中定义的方式进行。这种类型的设计模式属于行为型模式。

解决在多个子类中重复实现相同的方法的问题，通过将通用方法抽象到父类中来避免代码重复。

包含下面角色：

* **抽象父类（Abstract Class）**：
  * 定义了模板方法和一些抽象方法或具体方法。
* **具体子类（Concrete Classes）**：
  * 继承自抽象父类，并实现抽象方法。
* **钩子方法（Hook Method）（可选）**：
  * 在抽象父类中定义，可以被子类重写，以影响模板方法的行为。
* **客户端（Client）（可选）**：
  * 使用抽象父类和具体子类，无需关心模板方法的细节。

```cpp
#include <iostream>
#include <string>
#include <vector>

// 抽象基类，定义模板方法和具体步骤的接口
class DocumentProcessor {
public:
    virtual ~DocumentProcessor() = default;

    // 模板方法，定义了算法的框架
    virtual void  processDocument() final {
        loadDocument();
        processContent();
        saveDocument();
    }

protected:
    virtual void loadDocument() = 0;
    virtual void processContent() = 0;
    virtual void saveDocument() = 0;
};

// 具体子类，实现具体的步骤
class TextDocumentProcessor : public DocumentProcessor {
protected:
    void loadDocument() override {
        std::cout << "Loading text document...\n";
    }

    void processContent() override {
        std::cout << "Processing text content...\n";
    }

    void saveDocument() override {
        std::cout << "Saving text document...\n";
    }
};

class ImageDocumentProcessor : public DocumentProcessor {
protected:
    void loadDocument() override {
        std::cout << "Loading image document...\n";
    }

    void processContent() override {
        std::cout << "Processing image content...\n";
    }

    void saveDocument() override {
        std::cout << "Saving image document...\n";
    }
};

int main() {


    DocumentProcessor * processer = new TextDocumentProcessor();
    std::cout << "Processing text document:\n";
    processer->processDocument();
    delete processer;

    processer = new ImageDocumentProcessor();
    std::cout << "\nProcessing image document:\n";
    processer->processDocument();
    delete processer;

    return 0;
}
```

### 3.12 访问者模式（Visitor Pattern）

在访问者模式（Visitor Pattern）中，我们使用了一个访问者类，它改变了元素类的执行算法。通过这种方式，元素的执行算法可以随着访问者改变而改变。这种类型的设计模式属于行为型模式。根据模式，元素对象已接受访问者对象，这样访问者对象就可以处理元素对象上的操作。

包含的角色有：

* **访问者（Visitor）**：

  * 定义了访问元素的接口。
* **具体访问者（Concrete Visitor）**：

  * 实现访问者接口，提供对每个具体元素类的访问和相应操作。
* **元素（Element）**：

  * 定义了一个接受访问者的方法。
* **具体元素（Concrete Element）**：

  * 实现元素接口，提供一个`accept`方法，允许访问者访问并操作。
* **对象结构（Object Structure）（属于元素）**：

  * 定义了如何组装具体元素，如一个组合类。
* **客户端（Client）（可选）**：

  * 使用访问者模式对对象结构进行操作。

个人理解：Element中的`accept()`实际使用Visitor中的`visit()`，从而使Visitor决定Element算法。

```cpp
#include <iostream>
#include <vector>
#include <memory>
class ConcreteElementA;
class ConcreteElementB;
// 访问者接口
class Visitor {
public:
    virtual ~Visitor() = default;
    virtual void visit(class ConcreteElementA* element) = 0;
    virtual void visit(class ConcreteElementB* element) = 0;
};

// 具体访问者
class ConcreteVisitor1 : public Visitor {
public:
    void visit(ConcreteElementA* element) override {
        std::cout << "ConcreteVisitor1 visiting ConcreteElementA\n";
    }
    void visit(ConcreteElementB* element) override {
        std::cout << "ConcreteVisitor1 visiting ConcreteElementB\n";
    }
};

class ConcreteVisitor2 : public Visitor {
public:
    void visit(ConcreteElementA* element) override {
        std::cout << "ConcreteVisitor2 visiting ConcreteElementA\n";
    }
    void visit(ConcreteElementB* element) override {
        std::cout << "ConcreteVisitor2 visiting ConcreteElementB\n";
    }
};

// 元素接口
class Element {
public:
    virtual ~Element() = default;
    virtual void accept(Visitor* visitor) = 0;
};

// 具体元素
class ConcreteElementA : public Element {
public:
    void accept(Visitor* visitor) override {
        visitor->visit(this);  // 双分派：调用访问者的 visit 方法
    }
};

class ConcreteElementB : public Element {
public:
    void accept(Visitor* visitor) override {
        visitor->visit(this);
    }
};

// 对象结构
class ObjectStructure {
public:
    void addElement(std::shared_ptr<Element> element) {
        elements.push_back(element);
    }
    void accept(Visitor* visitor) {
        for (const auto& element : elements) {
            element->accept(visitor);
        }
    }
private:
    std::vector<std::shared_ptr<Element>> elements;
};

int main() {
    // 创建对象结构
    ObjectStructure os;
    os.addElement(std::make_shared<ConcreteElementA>());
    os.addElement(std::make_shared<ConcreteElementB>());

    // 创建访问者
    ConcreteVisitor1 visitor1;
    ConcreteVisitor2 visitor2;

    // 遍历对象结构并接受访问者
    std::cout << "Visiting with ConcreteVisitor1:\n";
    os.accept(&visitor1);

    std::cout << "\nVisiting with ConcreteVisitor2:\n";
    os.accept(&visitor2);

    return 0;
}
```
